class PlayerListCreateAPIView(generics.ListCreateAPIView):
    queryset=Player.objects.all()
    serializer_class=PlayerSerializer
    
class UpdatePlayer(generics.RetrieveUpdateDestroyAPIView):
    queryset=Player.objects.all()
    serializer_class=PlayerSerializer
    
class NpcAPIView(APIView):
    def get(self,request):
        npc=Npc.objects.all()
        serializer=NpcSerializer(npc,many=True)
        return Response(serializer.data)
    
    def post(self,request):
        serializer=NpcSerializer(data=request.data)
        if serializer.is_valid():
            serializer.save()
            return Response(serializer.data,status.HTTP_201_CREATED)
        return Response(serializer.errors,status.HTTP_400_BAD_REQUEST)
